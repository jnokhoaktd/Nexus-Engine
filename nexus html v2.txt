function toast(msg, isError = false) {
  const t = $("#toast");
  t.textContent = msg;
  t.style.borderColor = isError ? "var(--neg)" : "var(--primary)";
  t.style.display = "block";
  setTimeout(() => { t.style.display = "none"; }, 4000);
}

function updateApiStatus(connected) {
  isApiConnected = connected;
  const dot = $("#api-status");
  dot.className = "status-dot " + (connected ? "online" : "offline");
  dot.title = connected ? "API Online: Connected to " + API_BASE_URL : "API Offline: Running in local fallback mode";
}

async function apiCall(endpoint, options = {}) {
  try {
    const res = await fetch(API_BASE_URL + endpoint, options);
    if (!res.ok) throw new Error(`HTTP Error ${res.status}`);
    updateApiStatus(true);
    return await res.json();
  } catch (err) {
    updateApiStatus(false);
    throw err;
  }
}

// Check initial API availability

  try {
    const activeData = await apiCall('/datasets/active');
    if (activeData && activeData.source) {
      D = activeData;
      toast("Đã đồng bộ dữ liệu mới nhất từ API backend.");
    }
  } catch (err) {
    console.warn("Backend API không khả dụng, sử dụng bộ dữ liệu Offline Demo.", err);
  } finally {
    go(current);
  }
}

const insuf = (reason) => `<div class="insuf-box"><b>INSUFFICIENT DATA</b><div class="muted" style="margin-top:4px">${esc(reason)}</div></div>`;
const tag = (t) => `<span class="tag ${t === 'actual' ? 'actual' : 'demo'}">${esc(t)}</span>`;
const pill = (s) => `<span class="pill" style="background:${COL[s]}">${SENT[s]}</span>`;

function bars(items, color = "#6366f1", max = null, unit = "%") {
  const m = max ?? Math.max(...items.map(i => i.value), 1);
  return items.map(i => `<div class="bar-row"><span class="muted">${esc(i.label)}</span><div class="bar-track"><div class="bar-fill" style="width:${(i.value / m * 100).toFixed(1)}%;background:${i.color || color}"></div></div><span class="mono" style="text-align:right">${unit === "%" ? fmtPct(i.value) : i.value}</span></div>`).join("");
}

function stacked(items) {
  return items.map(i => {
    const t = (i.positive + i.neutral + i.negative) || 1;
    const seg = (k) => `<div style="width:${(i[k] / t * 100).toFixed(1)}%;background:${COL[k]}" title="${SENT[k]}: ${i[k]}"></div>`;
    return `<div class="stack-row"><span class="muted">${esc(i.label)}</span><div class="bar-track">${seg("positive")}${seg("neutral")}${seg("negative")}</div></div>`;
  }).join("");
}

function renderExplorer() {
  const sku = $("#fsku") ? $("#fsku").value : "", sent = $("#fsent") ? $("#fsent").value : "", q = $("#fq") ? $("#fq").value.trim().toLowerCase() : "";
  const rows = (D.explorer || []).filter(r => (!sku || r.sku === sku) && (!sent || r.absa === sent) && (!q || r.text.toLowerCase().includes(q) || r.sku.toLowerCase().includes(q)));
  const pageSize = 8;
  const pages = Math.max(1, Math.ceil(rows.length / pageSize));
  window.__page = Math.min(window.__page || 1, pages);
  const slice = rows.slice((window.__page - 1) * pageSize, window.__page * pageSize);
  
  const expEl = $("#explorer");
  if (!expEl) return;
  expEl.innerHTML = `
    <div style="overflow-x:auto"><table><thead><tr><th class="num">Dòng</th><th>SKU</th><th class="num">Sao</th><th>Nội dung</th><th>Cảm xúc</th><th>Khía cạnh</th></tr></thead><tbody>
    ${slice.map(r => `<tr><td class="num">${r.row}</td><td class="mono">${esc(r.sku)}</td><td class="num">${r.rating}</td><td class="wrap" style="max-width:420px">${esc(r.text)}</td><td>${pill(r.absa)}</td><td>${[...new Set(r.aspects.map(a => a.aspect))].map(a => `<span class="chip">${ASP[a]}</span>`).join("")}</td></tr>`).join("") || `<tr><td colspan="6" class="muted">Không có đánh giá khớp bộ lọc.</td></tr>`}

    <div class="pg"><span>${rows.length} đánh giá · trang ${window.__page || 1}/${pages}</span>
      <button class="btn" id="pprev">‹</button><button class="btn" id="pnext">›</button></div>`;
  $("#pprev").onclick = () => { window.__page = Math.max(1, (window.__page || 1) - 1); renderExplorer(); };
  $("#pnext").onclick = () => { window.__page = Math.min(pages, (window.__page || 1) + 1); renderExplorer(); };
}

function wireUnit() {
  const readInputs = () => Object.fromEntries([...document.querySelectorAll(".u-in")].map(i => [i.dataset.k, Number(i.value)]));
  const u = (x) => {
    const r = x.return_rate, kept = 1 - r;
    const discounts = x.selling_price * x.discount_rate, gad = x.selling_price - discounts, refunds = gad * r, netRev = gad - refunds;
    const cogs = x.cogs * kept, platform = netRev * x.platform_fee_rate, ful = x.fulfillment_cost * kept;
    const contrib = netRev - cogs - platform - x.ad_cost_per_order - ful;
    const rb = { reverse_shipping: r * x.reverse_shipping_cost, forward_shipping_nonrecovered: r * x.forward_shipping_nonrecovered, packaging_write_off: r * x.packaging_write_off, defect_loss: r * x.defect_loss };
    const rl = Object.values(rb).reduce((a, b) => a + b, 0);
    const net = contrib - rl - x.other_opex;
    const cap = netRev - cogs - platform - ful - rl - x.other_opex;
    const perRet = x.reverse_shipping_cost + x.forward_shipping_nonrecovered + x.packaging_write_off + x.defect_loss;
    const per = x.cogs * kept + x.ad_cost_per_order + x.fulfillment_cost * kept + r * perRet + x.other_opex;
    const be = per / (((1 - x.discount_rate) * kept * (1 - x.platform_fee_rate)) || 1);
    return { netRev, cogs, platform, ad: x.ad_cost_per_order, ful, contrib, rl, rb, net, margin: netRev ? net / netRev : 0, cap, roas: cap > 0 ? netRev / cap : null, be };
  };
  const show = () => {
    const x = readInputs(), r = u(x);
    const rows = [["Net Revenue", r.netRev], ["COGS (hàng giữ lại)", -r.cogs], ["Phí sàn", -r.platform], ["Quảng cáo", -r.ad], ["Fulfillment", -r.ful], ["Contribution", r.contrib], ["Return Loss", -r.rl], ["Net Profit", r.net]];
    $("#u-out").innerHTML = `<table><tbody>${rows.map(([a, b]) => `<tr><td>${a}</td><td class="num ${b < 0 ? "neg" : ""}">${fmtVnd(b)}</td></tr>`).join("")}</tbody></table>
      <div class="note">Biên ròng: <b>${fmtPct(r.margin * 100)}</b> · Giá hòa vốn: <b>${fmtVnd(r.be)}</b> · CAC tối đa/đơn: <b>${fmtVnd(r.cap)}</b> · ROAS hòa vốn: <b>${r.roas ? fmtNum(r.roas) : "—"}</b></div>`;
  };
  if ($("#u-calc")) $("#u-calc").onclick = show; show();
  if ($("#w-run")) $("#w-run").onclick = () => {
    const base = readInputs(), before = u(base);
    const after = u({ ...base, selling_price: Number($("#w-price").value), ad_cost_per_order: Number($("#w-ads").value) });
    const d = after.net - before.net;
    $("#w-out").innerHTML = `<div class="grid" style="margin-top:12px">
      <div class="card kpi"><div class="l">Net Profit trước</div><div class="v">${fmtVnd(before.net)}</div></div>
      <div class="card kpi"><div class="l">Net Profit sau</div><div class="v">${fmtVnd(after.net)}</div></div>
      <div class="card kpi"><div class="l">Chênh lệch</div><div class="v ${d >= 0 ? "pos" : "neg"}">${fmtVnd(d)}</div></div></div>`;
  };
}

async function fetchTelemetry() {
  try {
    const data = await apiCall('/system/telemetry');
    telemetryData = data;

    if ($("#t-ram")) $("#t-ram").textContent = data.ram + '%';

async function uploadFile(file) {
  const formData = new FormData();
  formData.append('file', file);
  try {
    toast("Đang tải file và chạy phân tích ABSA...");
    const result = await apiCall('/datasets/upload', { method: 'POST', body: formData });
    D = result;
    toast("Tải lên và xử lý file thành công!");
    go(current);
  } catch (err) {
    toast("Lỗi API upload: " + err.message + " (Đã quay về chế độ xem mẫu).", true);
  }
}

function wirePage(p) {
  if (telemetryTimer) { clearInterval(telemetryTimer); telemetryTimer = null; }
  
  if (p === "import") {

    if (dz && fi) {
      dz.onclick = () => fi.click();
      fi.onchange = (e) => { if (e.target.files.length) uploadFile(e.target.files[0]); };
      dz.ondragover = (e) => e.preventDefault();
      dz.ondrop = (e) => { e.preventDefault(); if (e.dataTransfer.files.length) uploadFile(e.dataTransfer.files[0]); };
    }
  }

  if (p === "customers") {
    ["#fsku","#fsent","#fq"].forEach(s => { if ($(s)) $(s).addEventListener("input", () => { window.__page = 1; renderExplorer(); }); });
    if ($("#dl-reviews")) { $("#dl-reviews").href = "data:text/csv;base64," + (D.csv.reviews || ''); $("#dl-reviews").download = "nexus_reviews.csv"; }
    window.__page = 1; renderExplorer();
  }

  if (p === "feedback") {
    if ($("#dl-aspects")) { $("#dl-aspects").href = "data:text/csv;base64," + (D.csv.aspects || ''); $("#dl-aspects").download = "nexus_aspects.csv"; }
  }

  if (p === "unit") wireUnit();

  if (p === "competitors") {
    if ($("#btn-import-comp")) {

        const fileInput = $("#comp-file-input");

        const formData = new FormData();

  }
}
